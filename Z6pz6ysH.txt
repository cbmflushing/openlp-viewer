    <script>
        // Fallback values used on first run or if localStorage is corrupt
        const DEFAULT_SETTINGS = {
            serverIp: 'localhost:4316',
            bgColor: '#006400',
            englishColor: '#ffffff',
            chineseColor: '#ffff00',
            footerColor: '#ffffff',
            englishFont: 'Arial, sans-serif',
            chineseFont: '"Microsoft YaHei", "SimHei", Arial, sans-serif',
            showTitle: false,
            ccliNumber: '',
            firstRun: true
        };
        
        let settings = { ...DEFAULT_SETTINGS };
        
        function loadSettings() {
            const saved = localStorage.getItem('cbmDisplaySettings');
            if (saved) {
                try {
                    settings = { ...DEFAULT_SETTINGS, ...JSON.parse(saved) };
                } catch (e) {
                    settings = { ...DEFAULT_SETTINGS };
                }
            }
            applySettings();
        }
        
        function saveSettings() {
            settings.serverIp = document.getElementById('server-ip').value.trim() || 'localhost:4316';
            settings.bgColor = document.getElementById('bg-color').value;
            settings.englishColor = document.getElementById('english-color').value;
            settings.chineseColor = document.getElementById('chinese-color').value;
            settings.footerColor = document.getElementById('footer-color').value;
            settings.englishFont = document.getElementById('english-font').value;
            settings.chineseFont = document.getElementById('chinese-font').value;
            settings.showTitle = document.getElementById('show-title').checked;
            settings.ccliNumber = document.getElementById('ccli-number').value.trim();
            settings.firstRun = false;
            
            localStorage.setItem('cbmDisplaySettings', JSON.stringify(settings));
            applySettings();
        }
        
        function applySettings() {
            document.body.style.setProperty('--bg-color', settings.bgColor);
            document.body.style.setProperty('--english-color', settings.englishColor);
            document.body.style.setProperty('--chinese-color', settings.chineseColor);
            document.body.style.setProperty('--footer-color', settings.footerColor);
            document.body.style.setProperty('--english-font', settings.englishFont);
            document.body.style.setProperty('--chinese-font', settings.chineseFont);
            
            document.getElementById('server-ip').value = settings.serverIp;
            document.getElementById('bg-color').value = settings.bgColor;
            document.getElementById('english-color').value = settings.englishColor;
            document.getElementById('chinese-color').value = settings.chineseColor;
            document.getElementById('footer-color').value = settings.footerColor;
            document.getElementById('english-font').value = settings.englishFont;
            document.getElementById('chinese-font').value = settings.chineseFont;
            document.getElementById('show-title').checked = settings.showTitle;
            document.getElementById('ccli-number').value = settings.ccliNumber;
            
            document.getElementById('title-display').style.display = settings.showTitle ? 'block' : 'none';
            
            updateFooter();
        }
        
        function updateFooter(ccliOverride) {
            let ccliNum = ccliOverride !== undefined ? ccliOverride : settings.ccliNumber;
            let footerHtml = 'CBM Flushing';
            if (ccliNum) {
                footerHtml += ' © CCLI# ' + ccliNum;
            }
            document.getElementById('footer').innerHTML = footerHtml;
        }
        
        function resetSettings() {
            settings = { ...DEFAULT_SETTINGS };
            localStorage.removeItem('cbmDisplaySettings');
            applySettings();
            showSettings();
        }
        
        function closeModal(save) {
            document.getElementById('modal-overlay').classList.add('hidden');
            settings.firstRun = false;
            localStorage.setItem('cbmDisplaySettings', JSON.stringify(settings));
            if (save) {
                saveSettings();
            }
        }
        
        function showSettings() {
            applySettings();
            document.getElementById('modal-overlay').classList.remove('hidden');
        }
        
        // Fetches the current live slide from OpenLP and updates the display.
        // OpenLP wraps Chinese text in <span> tags; English is the remaining plain text.
        // Song titles matching c### / h### / ch### are decoded as Chorus / Hymnary / Songbook.
        async function fetchAndDisplay() {
            try {
                const url = `http://${settings.serverIp}/api/controller/live/text`;
                const response = await fetch(url);
                
                if (!response.ok) {
                    throw new Error('Network response was not ok');
                }
                
                const data = await response.json();
                
                if (!data.results || !data.results.slides) {
                    document.getElementById('content').innerHTML = '';
                    return;
                }
                
                const slides = data.results.slides;
                let currentSlide = null;
                let songTitle = '';
                
                for (const slide of slides) {
                    if (slide.selected) {
                        currentSlide = slide;
                        songTitle = slide.title || '';
                        break;
                    }
                }
                
                if (!currentSlide || !currentSlide.text) {
                    document.getElementById('content').innerHTML = '';
                    document.getElementById('title-display').textContent = '';
                    updateFooter();
                    return;
                }
                
                if (settings.showTitle && songTitle) {
                    document.getElementById('title-display').textContent = songTitle;
                }
                
                let footerExtra = '';
                if (songTitle) {
                    const titleLower = songTitle.toLowerCase().trim();
                    const match = titleLower.match(/^([a-z]+)(\d+)/);
                    if (match) {
                        const prefix = match[1];
                        const number = match[2];
                        if (prefix === 'c') {
                            footerExtra = '<br>Chorus #' + number;
                        } else if (prefix === 'h') {
                            footerExtra = '<br>Hymnary #' + number;
                        } else if (prefix === 'ch') {
                            footerExtra = '<br>Songbook #' + number;
                        }
                    }
                }
                
                let footerHtml = 'CBM Flushing';
                if (settings.ccliNumber) {
                    footerHtml += ' © CCLI# ' + settings.ccliNumber;
                }
                footerHtml += footerExtra;
                document.getElementById('footer').innerHTML = footerHtml;
                
                let text = currentSlide.html || currentSlide.text;
                
                text = text.replace(/<br\s*\/?>/gi, '\n');
                
                // Chinese text arrives inside <span> tags; extract and deduplicate lines
                let chineseLinesSet = new Set();
                let chineseMatches = text.matchAll(/<span[^>]*>([\s\S]*?)<\/span>/gi);
                for (const match of chineseMatches) {
                    let chinesePart = match[1].trim();
                    chinesePart.split('\n').forEach(l => {
                        let cleaned = l.trim().replace(/\s+/g, '');
                        if (cleaned) chineseLinesSet.add(cleaned);
                    });
                }
                
                // Strip span tags to leave only English text
                let englishText = text.replace(/<span[^>]*>[\s\S]*?<\/span>/gi, '').trim();
                let englishLines = englishText.split('\n').map(l => l.trim()).filter(l => l.length > 0);
                
                let chineseLines = Array.from(chineseLinesSet);
                
                let htmlContent = '';
                
                for (const line of englishLines) {
                    let processed = line
                        .replace(/\s+/g, ' ')
                        .replace(/<sup>(.*?)<\/sup>/gi, '<sup>$1</sup>')
                        .replace(/<em>(.*?)<\/em>/gi, '<em>$1</em>')
                        .trim();
                    if (processed) {
                        htmlContent += `<p class="english">${processed}</p>`;
                    }
                }
                
                for (const line of chineseLines) {
                    let processed = line.replace(/\s+/g, '').trim();
                    if (processed) {
                        htmlContent += `<p class="chinese">${escapeHtml(processed)}</p>`;
                    }
                }
                
                document.getElementById('content').innerHTML = htmlContent;
                document.getElementById('error').style.display = 'none';
                
            } catch (error) {
                console.error('Error:', error);
                document.getElementById('error').textContent = 'Error: ' + error.message;
                document.getElementById('error').style.display = 'block';
            }
        }
        
        function escapeHtml(text) {
            const div = document.createElement('div');
            div.textContent = text;
            return div.innerHTML;
        }
        
        loadSettings();
        
        let isHidden = false;
        document.addEventListener('keydown', function(e) {
            if (e.key === '.' && !e.target.matches('input')) {
                isHidden = !isHidden;
                document.body.classList.toggle('hidden-content', isHidden);
            }
        });
        
        if (settings.firstRun) {
            document.getElementById('modal-overlay').classList.remove('hidden');
        }
        
        fetchAndDisplay();
        setInterval(fetchAndDisplay, 1000);
    </script>
